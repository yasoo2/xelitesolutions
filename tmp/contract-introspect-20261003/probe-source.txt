import { tools } from '../modules/tools/registry';
import * as fs from 'fs';
import * as path from 'path';

const NAMES = [
  'cloud_cost_estimator',
  'self_confidence_evaluator',
  'ask_user',
  'rss_fetch',
  'task_lifecycle',
  'read_file', // read control
  'write_file', // write control
  'bogus_tool_xyz_7788', // negative control
];

describe('contract-introspect probe', () => {
  test('input/output contract matrix', () => {
    const rows = NAMES.map((n) => {
      const def: any = (tools as any[]).find((t) => t && t.name === n) || null;
      if (!def) {
        return { name: n, registered: false };
      }
      const inp: any = def.inputSchema || null;
      const outp: any = def.outputSchema || null;
      const propKeys: string[] | null =
        inp && inp.properties && typeof inp.properties === 'object'
          ? Object.keys(inp.properties).sort()
          : null;
      const reqKeys: string[] | null =
        inp && Array.isArray(inp.required) ? [...inp.required].sort() : null;
      const outKeys: string[] | null =
        outp && outp.properties && typeof outp.properties === 'object'
          ? Object.keys(outp.properties).sort()
          : null;
      // Contract self-consistency: every required key must be a declared property.
      const requiredSubsetOfProperties =
        reqKeys === null || propKeys === null
          ? null
          : reqKeys.every((k) => propKeys.includes(k));
      return {
        name: n,
        registered: true,
        executeType: typeof def.execute,
        executeArity: typeof def.execute === 'function' ? def.execute.length : null,
        descriptionNonEmpty: typeof def.description === 'string' && def.description.length > 0,
        descriptionLen: typeof def.description === 'string' ? def.description.length : null,
        version: def.version ?? null,
        tags: Array.isArray(def.tags) ? def.tags : null,
        mockSupported: def.mockSupported ?? null,
        rateLimitPerMinute: def.rateLimitPerMinute ?? null,
        inputType: inp ? inp.type ?? null : null,
        inputProperties: propKeys,
        inputRequired: reqKeys,
        requiredSubsetOfProperties,
        outputType: outp ? outp.type ?? null : null,
        outputProperties: outKeys,
      };
    });
    const out = {
      museHead: 'aacc29f9',
      method: 'resolve-only; no tool executed; contract self-consistency = required keys subset of declared properties',
      rows,
    };
    const dir = path.resolve(__dirname, '../../../tmp/contract-introspect-20261003');
    fs.mkdirSync(dir, { recursive: true });
    fs.writeFileSync(path.join(dir, 'contracts.json'), JSON.stringify(out, null, 2));
    // eslint-disable-next-line no-console
    console.log(
      '[CONTRACT-INTROSPECT] ' +
        rows
          .map((r: any) =>
            r.registered
              ? `${r.name}:in=${r.inputType}:${JSON.stringify(r.inputProperties)}:req=${JSON.stringify(r.inputRequired)}:subset=${r.requiredSubsetOfProperties}:out=${r.outputType}:${JSON.stringify(r.outputProperties)}:arity=${r.executeArity}:desc=${r.descriptionLen}`
              : `${r.name}:UNREGISTERED`,
          )
          .join(' | '),
    );
    for (const r of (rows as any[]).slice(0, 7)) {
      expect(r.registered).toBe(true);
      expect(r.executeType).toBe('function');
      expect(r.descriptionNonEmpty).toBe(true);
      expect(r.inputType).toBe('object');
      expect(r.inputProperties).not.toBeNull();
      expect(r.requiredSubsetOfProperties).toBe(true);
      expect(r.outputType).not.toBeNull();
    }
    const bogus: any = rows[7];
    expect(bogus.registered).toBe(false);
  });
});
